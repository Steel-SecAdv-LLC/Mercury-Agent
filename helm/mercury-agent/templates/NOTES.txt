===========================================
API Server
===========================================

1. Get the API URL:
{{- if .Values.api.ingress.enabled }}
{{- range $host := .Values.api.ingress.hosts }}
   https://{{ $host.host }}
{{- end }}
{{- else if contains "NodePort" .Values.api.service.type }}
   export NODE_PORT=$(kubectl get --namespace {{ include "mercury-agent.namespace" . }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "mercury-agent.fullname" . }}-api)
   export NODE_IP=$(kubectl get nodes --namespace {{ include "mercury-agent.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.api.service.type }}
   NOTE: It may take a few minutes for the LoadBalancer IP to be available.
   Watch status: kubectl get svc --namespace {{ include "mercury-agent.namespace" . }} -w {{ include "mercury-agent.fullname" . }}-api
{{- else if contains "ClusterIP" .Values.api.service.type }}
   kubectl port-forward svc/{{ include "mercury-agent.fullname" . }}-api {{ .Values.api.service.port }}:{{ .Values.api.service.port }} -n {{ include "mercury-agent.namespace" . }}
   echo "API available at http://127.0.0.1:{{ .Values.api.service.port }}"
{{- end }}

2. Check API health:
   curl -s http://localhost:{{ .Values.api.service.port }}/health | jq

3. Access API documentation:
   http://localhost:{{ .Values.api.service.port }}/docs
{{- end }}

===========================================
Useful Commands
===========================================

# View pod status
kubectl get pods -n {{ include "mercury-agent.namespace" . }} -l app.kubernetes.io/name={{ include "mercury-agent.name" . }}

# View logs
kubectl logs -f deployment/{{ include "mercury-agent.fullname" . }}-api -n {{ include "mercury-agent.namespace" . }}

# Scale API replicas
kubectl scale deployment/{{ include "mercury-agent.fullname" . }}-api --replicas=5 -n {{ include "mercury-agent.namespace" . }}

# Check HPA status
kubectl get hpa -n {{ include "mercury-agent.namespace" . }}

# View events
kubectl get events -n {{ include "mercury-agent.namespace" . }} --sort-by='.lastTimestamp'
